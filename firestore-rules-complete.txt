rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null && request.auth.token.email == "YOUR_ADMIN_EMAIL_HERE";
    }
    match /publishedResults/{studentEmail} {
      match /reports/{reportId} {
        allow read: if request.auth != null && request.auth.token.email == studentEmail;
        allow write: if isAdmin();
      }
    }
    match /publishLog/{logId} {
      allow read, write: if isAdmin();
    }
    match /adminSettings/{settingId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }
    match /consultationRequests/{requestId} {
      allow create: if request.auth != null && request.resource.data.studentEmail == request.auth.token.email;
      allow read: if isAdmin() || (request.auth != null && resource.data.studentEmail == request.auth.token.email);
      allow update: if isAdmin();
    }
    match /reservedSlots/{slotId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null &&
        request.resource.data.keys().hasOnly(['start', 'end']) &&
        request.resource.data.start is string && request.resource.data.end is string;
      allow delete: if isAdmin();
    }
    match /adminWorkingState/{adminEmail} {
      allow read, write: if isAdmin() && request.auth.token.email == adminEmail;
    }
    match /mentorWorkingState/{mentorEmail} {
      allow read, write: if isAdmin() && request.auth.token.email == mentorEmail;
    }
    match /publishedMenteeData/{menteeEmail} {
      allow read: if request.auth != null && request.auth.token.email == menteeEmail;
      allow write: if isAdmin();
    }
    match /menteeOwnContent/{menteeEmail} {
      allow read: if isAdmin() || (request.auth != null && request.auth.token.email == menteeEmail);
      allow write: if request.auth != null && request.auth.token.email == menteeEmail;
    }
    match /studentGameSaves/{studentEmail} {
      allow read, write: if request.auth != null && request.auth.token.email == studentEmail;
    }
    match /gameQuestionBanks/{bankId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }
    match /gameEnrollments/{studentEmail} {
      allow read: if isAdmin() || (request.auth != null && request.auth.token.email.lower() == studentEmail);
      allow write: if isAdmin();
    }
  }
}
